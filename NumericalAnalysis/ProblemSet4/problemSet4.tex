        \section{Problem 1}
        \subsection{}
        We are deriving the 2-point open Newton-Coats formula
        \[
            \int_a^b f(x)dx\approx I_{2,\text{open}}(f) = \frac{b-a}{3}\bigg(2f(a+h) - f(a+2h) + 2f(a+3h)\bigg)
        \]
        for evenly spaced points with $h = \frac{b-a}{n+2}$.
        Now we set $n = 2$ so that $h=\frac{b-a}{4}$, and we need to sort out
        \begin{equation}
            I_{2,\text{open}}(f) = \int_a^b\sum_{i=0}^2 L_{2,i}(x)f(x_i) dx.
            \label{nc-equation}
        \end{equation}
        But we need to shift to an easier or simpler variable to work with.
        So we set $x=a+t\cdot h$, with $dx=h\cdot dt$.
        Now as $x$ moves from $a$ to $b$ we see that ($b=a+4h$), $t$ moves from $0$ to $4$, and we have
        \[
            \int_a^b f(x)dx = h\int_0^4 f(a+ht)dt,
        \]
        which we interpolate $f(a+ht)$ at $t=1,2,3$.
        We write out the Lagrangian polynomials $L_{2,0}, L_{2,1}, L_{2,2}$ for the sake of the problem.
        This follows as
        \begin{gather*}
            L_{2,0}(x) = \frac{(t-2)(t-3)}{(1-2)(1-3)} = \frac{(t-2)(t-3)}{2} = \frac{1}{2}(t^2 -5t + 6)\\
            L_{2,1}(x) = \frac{(t-1)(t-3)}{(2-1)(2-3)} = -(t-1)(t-3) = -t^2+ 4t - 3\\
            L_{2,2}(x) = \frac{(t-1)(t-2)}{(3-1)(3-2)} = \frac{(t-1)(t-2)}{2} = \frac{1}{2}(t^2 - 3t + 2).
        \end{gather*}
        And we then need to integrate these Lagrange interpolants, which we do as
        \begin{gather*}
            \int_0^4 \frac{1}{2}(t^2-5t+6)dt = \frac{t^3}{6} - \frac{5t^2}{4} + 3t \bigg|_0^4 = \frac{8}{3}\\
            \int_0^4 -t^2 + 4t - 3dt = -\frac{t^3}{3} + 2t^2 - 3t \bigg|_0^4 = -\frac{4}{3}\\
            \int_0^4 \frac{1}{2}(t^2 - 3t + 2)dt = \frac{t^3}{6} - \frac{3t^2}{2} + 2t \bigg|_0^4 = \frac{8}{3}
        \end{gather*}
        Thus, when we plug this into equation (\ref{nc-equation}), we get the following
        \begin{eqnarray*}
            \int_a^b f(x)dx & \approx & h\bigg(\frac{8}{3}f(a+h) - \frac{4}{3}f(a+2h) + \frac{8}{3}f(a+3h)\bigg)\\
             & \approx & \frac{4h}{3}\bigg(2f(a+h) - f(a+2h) + 2f(a+3h)\bigg)\\
             & \approx & \frac{b-a}{3}\bigg(2f(a+h) - f(a+2h) + 2f(a+3h)\bigg).
        \end{eqnarray*}
        which solves our problem

        \section{Problem 2}
        Let
        \[
            I(f) = \int_a^b f(x)dx
        \]
        and let $[a,b]$ be divided into subintervals in the following fashion
        \[
            \big\{[a,x_1+h],[x_2-h,x_2+h],\ldots,[x_n-h,b]\big\},
        \]
        with $x_1,\ldots,x_n$ being the midpoints of the subintervals.
        We thusly see that $h=\frac{b-a}{2n}$, that $x_j=a + (2j-1)h$, and that each of our subintervals has
        width $2h$.
        Now consider the $j^{th}$ subinterval $[x_j-h,x_j+h]$.
        We can write $f(x)$ over the interval using Taylor's theorem as
        \[
            f(x) = f(x_j) + f'(x_j)(x-x_j) + \frac{f''(\eta_x)}{2}(x-x_j)^2,
        \]
        for some $\eta_x$ between $x$ and $x_j$.
        So we integrate
        \begin{eqnarray*}
            \int_{x_j-h}^{x_j+h} f(x)dx & = & \int_{x_j-h}^{x_j+h}\left(f(x_j) + f'(x_j)(x-x_j) + \frac{f''(\eta_x)}{2}(x-x_j)^2 \right)dx\\
             & = & \int_{x_j-h}^{x_j+h}f(x_j)dx  + \int_{x_j-h}^{x_j+h}f'(x_j)(x-x_j)dx + \int_{x_j-h}^{x_j+h}\frac{f''(\eta_x)}{2}(x-x_j)^2 dx\\
             & = & 2hf(x_j) + 0 + \frac{f''(\xi_j)}{2}\int_{x_j-h}^{x_j+h} (x-x_j)^2 dx,
        \end{eqnarray*}
        for some $\xi_j$ clearly dependent upon our choice of $\eta_x$.
        But notice that our last integral can be manipulated by a transformation of variables in the following
        fashion
        \[
            \int_{x_j-h}^{x_j+h}(x-x_j)^2 dx = \int_{-h}^h t^2 dt = \frac{2h^3}{3}.
        \]
        Substituting that back in to the over all equation we see that
        \[
            \int_{x_j-h}^{x_j+h} f(x)dx + \frac{f''(\xi_j)}{\cancel{2}}\frac{\cancel{2}h^3}{3}.
        \]
        So we take this result and sum across all the subintervals to get that
        \[
            \int_b^a f(x)dx = \sum_{j=1}^n\int_{x_j-h}^{x_j+h}f(x)dx = 2h\sum_{j=1}^n f(x_j) + \frac{h^3}{3}\sum_{j=1}^n f''(\xi_j).
        \]
        But we notice that $\sum_{j=1}^n f''(\xi_j) = nf''(\xi)$ with $f''(\xi)$ representing the average of all $f''(\xi_j)$.
        And with some careful substitution, along with noticing that $n=\frac{b-a}{2h}$ by manipulating our definition of $h$,
        we see that
        \[
            \int_a^b f(x)dx = 2h\sum_{j=1}^n f(x_j) + \frac{(b-a)h^2}{6}f''(\xi),
        \]
        where $\xi\in[a,b]$, and we are done.
