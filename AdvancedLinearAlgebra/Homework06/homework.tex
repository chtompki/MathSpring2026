        % P.4.9
        \noindent\textbf{P.4.9.} If $A\in\mathbf{M}_n$ is to be represented as $A=XY^\top$ for some
        $X,Y\in\mathbf{M}_{n\times r}$, explain why $r$ cannot be smaller than $\text{rank}(A)$.
        \begin{solution}
            We claim that the columns of $A$ must lie in the column space of $X$.
            To see this we write
            \begin{displaymath}
                Y^{\top} = \left[\begin{matrix}
                                     y_1^{\top}\\
                                     y_2^{\top}\\
                                     \vdots\\
                                     y_r^{\top}
                \end{matrix}\right],
            \end{displaymath}
            where $y_i\in\mathbb{F}^n$ are columns of $Y$. Then we have that
            \begin{displaymath}
                XY^{\top} = \sum_{k=1}^r x_k y_k^{\top}.
            \end{displaymath}
            Now choose an arbitrary vector $v\in\mathbb{F}^n$,
            \begin{displaymath}
                Av = XY^{\top}v = X(Y^\top v)
            \end{displaymath}
            with $Y^\top v\in\mathbb{F}^r$.
            So $Av$ is a linear combination of the $r$ columns of $X$, which gives us
            $\text{col}(A)\subseteq\text{col}(X)$.
            So
            \begin{displaymath}
                \text{rank}(A) = \text{dim}(\text{col}(A)) \le \text{dim}(\text{col}(X)) = \text{rank}(X)\le r
            \end{displaymath}
            So $\text{rank}(A)\le r$, and we are done.
        \end{solution}

        % P.4.10
        \noindent\textbf{P.4.10.} For $X\in\mathbf{M}_{m\times n}$, $\nu(X)$ denote the nullity of $X$.
        \begin{enumerate}[label=(\alph*)]
            \item Show that $\nu(X^\top) = \nu(X) + m - n$.
            \item Let $A\in\mathbf{M}_{m\times k}$ and $B\in\mathbf{M}_{k\times n}$.
                  Show that the rank inequality given by
                \begin{displaymath}
                    \text{rank}(A) + \text{rank}(B) - k \le \text{rank}(AB)
                \end{displaymath}
                  is equivalent to
                \begin{equation}
                    \nu(AB) \le \nu(A) + \nu(B).
                    \label{p410one}
                \end{equation}
            \item If $A,B\in\mathbf{M}_{n}$, show that the rank inequality given by
            \begin{displaymath}
                \text{rank}(AB) \le \min\{\text{rank}(A), \text{rank}(B)\}.
            \end{displaymath}
            is equivalent to
            \begin{equation}
                \max\{\nu(A),\nu(B)\}\le\nu(AB)
                \label{p410two}
            \end{equation}
            The inequalities (\ref{p410one}) and (\ref{p410two}) are known as \emph{Sylvester's law of nullity}.
            \item Consider $A = [\begin{matrix}0 & 1\end{matrix}]$ and $B = [\begin{matrix}1 & 0\end{matrix}]^\top$.
                  Show that the inequality (\ref{p410two}) need not be valid if the matrices are not square.
        \end{enumerate}
        \begin{proof}
            (a) By the rank-nullity theorem
            \begin{eqnarray*}
                \text{rank}(X) + \nu(X) & = & n\\
                \text{rank}(X) & = & n - \nu(X)\\
                \text{rank}(X^\top) + \nu(X^\top) & = & m\\
                \text{rank}(X^\top) & = & m - \nu(X^\top)\\
            \end{eqnarray*}
            Yet we know that $\text{rank}(X) = \text{rank}(X^\top)$.
            So we have:
            \begin{displaymath}
                \nu(X^\top) = \nu(X) + m - n.
            \end{displaymath}

            (b) Similar to (a) we write out the rank equations for the matrices
            \begin{equation}
                \begin{aligned}
                    \text{rank}(A) & = k - \nu(A)\\
                    \text{rank}(B) & = n - \nu(B)\\
                    \text{rank}(AB) & = n - \nu(AB).
                \end{aligned}
                \label{410b}
            \end{equation}
            Substituting into $\text{rank}(A) + \text{rank}(B) - k \le \text{rank}(AB)$ we
            get
            \begin{eqnarray*}
                \text{rank}(A) + \text{rank}(B) - k & \le & \text{rank}(AB)\\
                (k-\nu(A)) + (n-\nu(B)) - k & \le & n-\nu(AB) \\
                n - \nu(A) - \nu(B) & \le & n \nu(AB)\\
                \nu(A) + \nu(B) & \ge & \nu(AB),
            \end{eqnarray*}
            where in that last step we both subtracted both sides by $n$ and
            multiplied both sides of the inequality by $-1$ switching its direction.
            And we are done.

            (c) Similarly to (b) let's consider equation (\ref{410b}).
            If we consider what $\text{rank}(AB) \le \min\{\text{rank}(A), \text{rank}(B)\}$ is actually saying
            we have that we have both $\text{rank}(AB) \le \text{rank}(A)$ and $\text{rank}(AB) \le \text{rank}(B)$.
            We see that
            \begin{gather*}
                n - \nu(AB) \le n - \nu(A) \qquad\Longleftrightarrow\qquad \nu(A)\le\nu(AB), \\
                n - \nu(AB) \le n - \nu(B) \qquad\Longleftrightarrow\qquad \nu(B)\le\nu(AB). \\
            \end{gather*}
            So we have
            \begin{displaymath}
                \max\{\nu(A),\nu(B)\}\le\nu(AB)
            \end{displaymath}

            (d) We compute $AB$ and $BA$ for the sake of discussion
            \begin{displaymath}
                AB = \left[\begin{matrix}
                               0 & 1
                \end{matrix}\right]
                \left[\begin{matrix}
                          1 \\ 0
                \end{matrix}\right] = 0
                \quad\text{and}\quad
                BA = \left[\begin{matrix}
                               1 \\ 0
                \end{matrix}\right]
                \left[\begin{matrix}
                          0 & 1
                \end{matrix}\right] =
                \left[\begin{matrix}
                          0 & 1\\
                          0 & 0
                \end{matrix}\right]
            \end{displaymath}
            We also look at what the null spaces of the respective matrices are.
            So we describe our linear transformations as functions
            \begin{displaymath}
                A = \left[\begin{matrix}
                              0 & 1
                \end{matrix}\right]
                = (x,y)\longmapsto y
                \quad\text{and}\quad
                B = \left[\begin{matrix}
                              1 \\ 0
                \end{matrix}\right] =
                \hat{x}\longmapsto
                \left[\begin{matrix}
                          \hat{x} & \hat{0}
                \end{matrix}\right],
            \end{displaymath}
            where $\hat{x} = [\begin{matrix}x_1 & \cdots & x_n\end{matrix}]$ and $\hat{0}$ is the $n$-ary
            zero vector.
            Given that $\text{ker}(A) = \{(x,0): x\in\mathbb{F}\}$ we have that $\nu(A)=1$.
            Also, notice $\text{ker}(B) = \{\hat{0}\}$, so $\nu(B)=0$.
            Lastly notice that both $\nu(AB)=1$ by $AB$ being the zero map, and $\nu(BA)=1$ by
            single element linear transformation.
            But then we have $\max\{0,1\}\le 1$ in either case which does not help us. Hm.

            So we consider the following counterexample
            \begin{displaymath}
                A = \left[\begin{matrix}
                              1 & 0 & 0\\
                              0 & 1 & 0
                \end{matrix}\right]
                \quad\text{and}\quad
                B = \left[\begin{matrix}
                              1 & 0\\
                              0 & 1\\
                              0 & 0
                \end{matrix}\right].
            \end{displaymath}
            So,
            \begin{displaymath}
                AB = \left[\begin{matrix}
                              1 & 0 & 0\\
                              0 & 1 & 0
                \end{matrix}\right]
                \left[\begin{matrix}
                          1 & 0\\
                          0 & 1\\
                          0 & 0
                \end{matrix}\right]
                = \left[\begin{matrix}
                            1 & 0\\
                            0 & 1\\
                \end{matrix}\right].
            \end{displaymath}
            The nullity of $A$ is $1$ because our rank is 2.
            The nullity of $B$ is $0$,
            and finally the nullity of $AB$ is $0$ because it's invertible.
            So we truly are stuck with
            $\max\{0,1\}\le 0$ which doesn't hold.
        \end{proof}

        % P.4.13
        \noindent\textbf{P.4.13.} Let $\mathcal{V}$ be a nonzero $n$-dimensional vector space over the field
        $\mathbb{F}$, and let $T\in\mathfrak{L}(\mathcal{V},\mathbb{F})$ be a nonzero linear transformation.
        Why is $\text{dim}(\text{ker}(T)) = n-1$?
        \begin{proof}
            Let $T:\mathcal\rightarrow\mathbb{F}$ be a linear transformation.
            Now the $\text{rank}(T)\le 1$, and because $T$ is nonzero, we must have
            $\text{rank}(T) = 1$.
            By the rank nullity theorem we see that
            \begin{eqnarray*}
                1 + \nu(T) & = & n \\
                \nu(T) & = & n - 1.
            \end{eqnarray*}
            Done.
        \end{proof}

                \end{matrix}\right]
            \end{displaymath}
            Finally, we perform $r_1:r_1-2r_3$, and $r1:r1+r2$

                          1 & 0 & 0 & 1 & -2\\
