        % Problem P.1.2
        \noindent\textbf{P.1.2} Is $[\begin{matrix}1 & 0 & 1\end{matrix}]^\top \in\mathbb{R}^3$ in the span of
        $[\begin{matrix}-1 & 3 & 2\end{matrix}]^\top$ and $[\begin{matrix} 1 & 1 & -2\end{matrix}]^\top$? Why?
        \begin{solution}
            We will show that $[\begin{matrix}1 & 0 & 1\end{matrix}]^\top
            \not\subseteq \text{span}([\begin{matrix}-1 & 3 & 2\end{matrix}]^\top$, $[\begin{matrix} 1 & 1 & -2\end{matrix}]^\top)$
            by contradiction. So assume that there is some $a,b\in\mathbb{R}$ with
            \begin{displaymath}
                a\left[\begin{matrix}
                           -1 \\
                           3 \\
                           2
                \end{matrix}\right] +
                b\left[\begin{matrix}
                          1 \\
                          1 \\
                          -2
                \end{matrix}\right] =
                \left[\begin{matrix}
                          1 \\
                          0 \\
                          1
                \end{matrix}\right]
            \end{displaymath}
            Which if we expand we get the following equations:
            \begin{eqnarray*}
                -a + b & = & 1 \\
                3a + b & = & 0 \\
                2a - 2b & = & 1
            \end{eqnarray*}
            If we simply try to solve the first and third equation we need that the matrix equation to hold:
            \begin{displaymath}
                \left[\begin{matrix}
                          -1 & 1\\
                          2 & -2
                \end{matrix}\right]
                \left[\begin{matrix}
                          a \\
                          b
                \end{matrix}\right] =
                \left[\begin{matrix}
                          a \\
                          b
                \end{matrix}\right]
            \end{displaymath}
            For this to be the case we would need that $\left[\begin{matrix} -1 & 1\\ 2 & -2 \end{matrix}\right]$ be
            invertible so its determinant would need to be non-zero. Let's calculate the determinant:
            \begin{displaymath}
                \text{det}\left[\begin{matrix}
                                    -1 & 1\\
                                    2 & -2
                \end{matrix}\right] = (2 - 2) = 0.
            \end{displaymath}
            So our matrix is not invertible, and we cannot find a unique $a$ and $b$ to solve our system of equations.
            $[\begin{matrix}1 & 0 & 1\end{matrix}]^\top
            \not\subseteq \text{span}([\begin{matrix}-1 & 3 & 2\end{matrix}]^\top$, $[\begin{matrix} 1 & 1 & -2\end{matrix}]^\top)$
        \end{solution}

        % Problem P.1.3
        \noindent\textbf{P.1.3} Is $[\begin{matrix}7 & 8 & 9\end{matrix}]^\top \in\mathbb{R}^3$ in the span of
        $[\begin{matrix}1 & 2 & 3\end{matrix}]^\top$ and $[\begin{matrix}4 & 5 & 6\end{matrix}]^\top$? Why? Is
        $[\begin{matrix}10 & 11 & 12\end{matrix}]^\top$?
        \begin{solution}
            In a similar fashion to problem \textbf{P.1.2}, we choose an $a,b\in\mathbb{R}$, and we set:
            \begin{displaymath}
                a\left[\begin{matrix}
                           1 \\
                           2 \\
                           3
                \end{matrix}\right] +
                b\left[\begin{matrix}
                           4 \\
                           5 \\
                           6
                \end{matrix}\right] =
                \left[\begin{matrix}
                          7 \\
                          8 \\
                          9
                \end{matrix}\right].
            \end{displaymath}
            This reduces to the equations:
            \begin{eqnarray*}
                a + 4b & = & 7\\
                2a + 5b & = & 8 \\
                3a + 6b & = & 9 \\
            \end{eqnarray*}
            So if we let $a = -4b + 7$, as given by the first equation and then substitute into the second equation,
            we see that
            \begin{eqnarray*}
                2(-4b + 7) + 5b & = & 8 \\
                -8b + 14 + 5b & = & 8 \\
                -3b & = & -6 \\
                b & = & 2
            \end{eqnarray*}
            Substituting back into our top equality we see that $a + 4(2) = 7$, thus $a = -1$. Furthermore if we plug these
            values back into our original linear combination, we get that
            \begin{eqnarray*}
                -1\left[\begin{matrix}
                           1 \\
                           2 \\
                           3
                \end{matrix}\right] +
                2\left[\begin{matrix}
                           4 \\
                           5 \\
                           6
                \end{matrix}\right] & = &
                \left[\begin{matrix}
                          7 \\
                          8 \\
                          9
                \end{matrix}\right] \\
                \left[\begin{matrix}
                            -1 \\
                            -2 \\
                            -3
                \end{matrix}\right] +
                \left[\begin{matrix}
                           8 \\
                           10 \\
                           12
                \end{matrix}\right] & = &
                \left[\begin{matrix}
                          7 \\
                          8 \\
                          9
                \end{matrix}\right]
            \end{eqnarray*}
            If we were to do the same inspection for the vector $[\begin{matrix}10 & 11 & 12\end{matrix}]^\top$ we write
            $a,b\in\mathbb{R}$ with
            \begin{displaymath}
                a\left[\begin{matrix}
                           1 \\
                           2 \\
                           3
                \end{matrix}\right] +
                b\left[\begin{matrix}
                           4 \\
                           5 \\
                           6
                \end{matrix}\right] =
                \left[\begin{matrix}
                          10 \\
                          11 \\
                          12
                \end{matrix}\right],
            \end{displaymath}
            We then expand to:
            \begin{eqnarray*}
                a + 4b & = & 10\\
                2a + 5b & = & 11 \\
                3a + 6b & = & 12 \\
            \end{eqnarray*}
            So we have that $a = -4b + 10$ which if we plug into the second equation we get:
            \begin{eqnarray*}
                2(-4b + 10) + 5b & = & 11 \\
                -8b + 20 + 5b & = & 11 \\
                -3b & = & -9 \\
                b & = & 6
            \end{eqnarray*}
            And if we plug $b$ back in we get $a = -24 + 10 = -14$. So we test this out to see if it works:
            \begin{eqnarray*}
                -14\left[\begin{matrix}
                           1 \\
                           2 \\
                           3
                \end{matrix}\right] +
                6\left[\begin{matrix}
                           4 \\
                           5 \\
                           6
                \end{matrix}\right] & = &
                \left[\begin{matrix}
                          10 \\
                          11 \\
                          12
                \end{matrix}\right]\\
                \left[\begin{matrix}
                             -14 \\
                             -28 \\
                             -42
                \end{matrix}\right] +
                \left[\begin{matrix}
                           24 \\
                           30 \\
                           36
                \end{matrix}\right] & = &
                \left[\begin{matrix}
                          10 \\
                          2 \\
                          -6
                \end{matrix}\right] \neq \left[\begin{matrix}
                                                                   10 \\
                                                                   11\\
                                                                   12
                \end{matrix}\right]
            \end{eqnarray*}
            So $[\begin{matrix}10 & 11 & 12\end{matrix}]^\top
            \not\subseteq \text{span}([\begin{matrix}1 & 2 & 3\end{matrix}]^\top$, $[\begin{matrix} 4 & 5 & 6\end{matrix}]^\top)$
        \end{solution}

        % Problem P.1.8
        \noindent\textbf{P.1.8} Consider the followoing lists of vectors in $\mathbb{R}^3:$
        \begin{displaymath}
            \beta = \left[\begin{matrix} 1 \\ 2 \\ 3 \end{matrix}\right],
            \left[\begin{matrix}4 \\ s\\ 6\end{matrix}\right],
            \left[\begin{matrix}7 \\ 8\\ 9\end{matrix}\right],
        \end{displaymath}
        and
        \begin{displaymath}
            \gamma = \left[\begin{matrix} 4 \\ 2 \\ 6 \end{matrix}\right],
            \left[\begin{matrix}1 \\ t\\ -1\end{matrix}\right],
            \left[\begin{matrix}2 \\ -2\\ 3\end{matrix}\right],
        \end{displaymath}
        For what values of $s$ is $\beta$ linearly dependent. For what values of $t$
        is $\gamma$ linearly dependent.
        \begin{solution}
            ($\beta$) We know that $\beta$ is linearly dependent when $\text{det}(M_\beta) = 0$, where $M_\beta$
            represents the matrix with the vectors of $\beta$ as its columns. So let us compute the
            following:
            \begin{eqnarray*}
                \text{det}\left[\begin{matrix}
                                    1 & 4 & 7 \\
                                    2 & s & 8 \\
                                    3 & 6 & 9
                \end{matrix}\right] & = & 0 \\
                1\cdot\text{det}\left[\begin{matrix}
                                     s & 8\\
                                     6 & 9
                \end{matrix}\right] +
                4\cdot\text{det}\left[\begin{matrix}
                                     2 & 8\\
                                     3 & 9
                \end{matrix}\right] +
                7\cdot\text{det}\left[\begin{matrix}
                                     2 & s\\
                                     3 & 6
                \end{matrix}\right] & = & 0\\
                1(9s - 48) + 4(18 -24) + 7(12 - 3s) & = & 0 \\
                9s - 48 + 72 - 96 + 84 - 21s & = & 0 \\
                -12s + 12 & = & 0 \\
                s & = & 1
            \end{eqnarray*}
            So when $s = 1$ $\beta$ is linearly dependent.

            ($\gamma$) Let $M_\gamma$ be the matrix comprised of the vectors of $\gamma$ as columns. Similarly to above,
            we attempt to compute $\text{det}(M_\gamma) = 0$. So we expand:
            \begin{eqnarray*}
                \text{det}\left[\begin{matrix}
                                    4 & 1 & 2 \\
                                    2 & t & -2 \\
                                    6 & -1 & 3
                \end{matrix}\right] & = & 0 \\
                4\cdot\text{det}\left[\begin{matrix}
                                          t & -2\\
                                          -1 & 3
                \end{matrix}\right] +
                1\cdot\text{det}\left[\begin{matrix}
                                          2 & -2\\
                                          6 & 3
                \end{matrix}\right] +
                2\cdot\text{det}\left[\begin{matrix}
                                          2 & t\\
                                          6 & -1
                \end{matrix}\right] & = & 0\\
                4(3t - 2) + (6 + 12) + 2(-2 - 6t) & = & 0 \\
                12t - 8 + 6 + 12 - 4 - 12t & = & 0 \\
                6 & \neq & 0
            \end{eqnarray*}
            Because our terms $12t$ and $-12t$ cancel out and we get that $\text{det}(M_\gamma) = 6$, we have that
            $\gamma$ is linearly independent for any value of $t$.
        \end{solution}

        % Problem P.1.12
        \noindent\textbf{P.1.12} Show that the set of $n\times n$ Hermitian matricies is a real vector space.
        \begin{proof}
            Let $\mathcal{H}_n\subseteq\mathbf{M}_n(\mathbb{C})$ be the Hermitian. So by definition for
            $A\in\mathcal{H}_n$, we have that $A^* = A$. Now we need to prove the vector space axioms using
            $\mathbb{R}$ as our set of scalars. Begin by setting $A,B\in\mathcal{H}_n$ and $c\in\mathbb{R}$.

            \emph{Closure under addition.} We simply perform some easy algebraic manipulation beginning with the
            conjugate transpose of the addition $A + B$
            \begin{displaymath}
                (A + B)^* = A^* + B^* = A + B \in\mathcal{H}_n.
            \end{displaymath}

            \emph{Scalar multiplication.} For this vector space axiom, we begin with $(cA)^*$ and perform algebraic
            manipulation as follows:
            \begin{displaymath}
                (cA)^* = \bar{c}A^* = cA^* = cA \in\mathcal{H}_n
            \end{displaymath}
            since $\bar{c} = c$ by $c\in\mathbb{R}$.

            \emph{The zero vector.} This is trivially true as
            \begin{displaymath}
                \hat{0}^* = \hat{0} \in\mathcal{H}_n
            \end{displaymath}

            \emph{Note.} Notice $\mathcal{H_n}$ is not a complex vector space because $(iA)^* = -iA^* = -iA\neq iA$.
        \end{proof}
